const getToolingCostPerPartForBreak = (breakQty: number) => {
  return toolingRows.reduce((sum, row) => {
    const total = Number(row.totalCost ?? 0);

    if (row.pricingTreatment === "BLENDED_UNIT_PRICE") {
      return sum + (breakQty > 0 ? total / breakQty : 0);
    }

    if (row.pricingTreatment === "AMORTIZED") {
      const amortQty = Number(row.amortizationQty || 0);
      return sum + (amortQty > 0 ? total / amortQty : 0);
    }

    return sum;
  }, 0);
};

const getAdjustmentCostForPartAndBreak = (
  partId: string | undefined,
  breakQty: number,
  materialCostPerPart: number,
  laborCostPerPart: number
) => {
  const applicable = adjustmentRows.filter((row) => {
    if (!row.includeInCustomerPrice) return false;
    if (row.pricingMode === "INTERNAL_ONLY") return false;
    if (row.appliesToScope === "RFQ") return true;
    return row.appliesToScope === "PART" && row.rfqPartId === partId;
  });

  return applicable.reduce((sum, row) => {
    if (row.pricingMode === "FLAT") {
      return sum + (breakQty > 0 ? Number(row.amount || 0) / breakQty : 0);
    }

    if (row.pricingMode === "PER_PART") {
      return sum + Number(row.amount || 0);
    }

    if (row.pricingMode === "PERCENT_OF_MATERIAL") {
      return sum + materialCostPerPart * (Number(row.percentValue || 0) / 100);
    }

    if (row.pricingMode === "PERCENT_OF_LABOR") {
      return sum + laborCostPerPart * (Number(row.percentValue || 0) / 100);
    }

    return sum;
  }, 0);
};

const getShippingCostPerPartForBreak = (partId: string | undefined, breakQty: number) => {
  const applicable = shippingRows.filter((row) => {
    if (!row.includeInCustomerPrice) return false;
    if (row.shippingMode === "INTERNAL_ONLY") return false;
    if (!row.rfqPartId) return true;
    return row.rfqPartId === partId;
  });

  return applicable.reduce((sum, row) => {
    if (row.shippingMode === "PER_PART") {
      return sum + Number(row.amount || 0);
    }

    if (row.shippingMode === "PER_PO") {
      return sum + (breakQty > 0 ? Number(row.amount || 0) / breakQty : 0);
    }

    return sum;
  }, 0);
};

const getMaterialCostPerPartForPart = (partId: string | undefined) => {
  return bomLines
    .filter((line) => line.rfqPartId === partId)
    .reduce((sum, row) => {
      const qty = Number(row.quantityPerPart || 0);
      const unit = Number(row.estimatedUnitCost || 0);
      const scrap = Number(row.scrapPercent || 0) / 100;
      return sum + qty * unit * (1 + scrap);
    }, 0);
};

const getLaborCostPerPartForPartAndBreak = (partId: string | undefined, breakQty: number) => {
  const rows = processRows.filter((row) => row.rfqPartId === partId);

  const setupCost = rows.reduce(
    (sum, row) => sum + Number(row.setupHours || 0) * Number(row.hourlyRate || 0),
    0
  );

  const recurringPerPart = rows.reduce(
    (sum, row) => sum + Number(row.hoursPerPart || 0) * Number(row.hourlyRate || 0),
    0
  );

  return recurringPerPart + (breakQty > 0 ? setupCost / breakQty : 0);
};
Compute pricing matrix
const pricingMatrix = useMemo(() => {
  return quantityBreakRows
    .slice()
    .sort((a, b) => a.sortOrder - b.sortOrder)
    .map((qb) => {
      const breakQty = Number(qb.quantity || 0);
      const toolingPerPart = getToolingCostPerPartForBreak(breakQty);

      const partRows = parts
        .filter((p) => p.id)
        .map((part) => {
          const materialPerPart = getMaterialCostPerPartForPart(part.id);
          const laborPerPart = getLaborCostPerPartForPartAndBreak(part.id, breakQty);
          const adjustmentPerPart = getAdjustmentCostForPartAndBreak(
            part.id,
            breakQty,
            materialPerPart,
            laborPerPart
          );
          const shippingPerPart = getShippingCostPerPartForBreak(part.id, breakQty);

          const totalCostPerPart =
            materialPerPart +
            laborPerPart +
            toolingPerPart +
            adjustmentPerPart +
            shippingPerPart;

          const marginDecimal = Number(marginPercent || 0) / 100;
          const sellPricePerPart =
            marginDecimal >= 1
              ? totalCostPerPart
              : totalCostPerPart / (1 - marginDecimal);

          const extendedPrice = sellPricePerPart * breakQty;

          return {
            partId: part.id,
            partNumber: part.partNumber,
            materialPerPart,
            laborPerPart,
            toolingPerPart,
            adjustmentPerPart,
            shippingPerPart,
            totalCostPerPart,
            sellPricePerPart,
            extendedPrice,
          };
        });

      return {
        id: qb.id,
        label: qb.label,
        quantity: breakQty,
        rows: partRows,
      };
    });
}, [
  quantityBreakRows,
  parts,
  bomLines,
  processRows,
  toolingRows,
  adjustmentRows,
  shippingRows,
  marginPercent,
]);

  WHERE epochEmployeeId = $1
  ORDER BY punchTime DESC
  LIMIT 1
`, [req.user.id]);

if (type === 'clock_in' && lastType === 'clock_in') {
  return res.status(400).json({ error: 'Already clocked in' });

  return res.status(400).json({ error: 'Must clock in first' });

router.get('/status', authenticateToken, async (req, res) => {
  const last = await db.query(`
    SELECT punchType, punchTime
    FROM punch_events
    WHERE epochEmployeeId = $1
    ORDER BY punchTime DESC
    LIMIT 1
  `, [req.user.id]);

  res.json({
    status: last[0]?.punchType || 'clock_out',
    lastPunch: last[0] || null
  });
});

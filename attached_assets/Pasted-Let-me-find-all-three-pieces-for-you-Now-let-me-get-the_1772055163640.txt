Item selection — how selectedItems is built in the parent (QCShippingQueuePage.tsx lines 212–279):

const transformedShipmentItems = useMemo(() => {
  // Builds lookup maps from poOrders data (customer → PO → items)
  const orderIdMap = new Map<string, any>();
  const compositeKeyMap = new Map<string, any>();
  (poOrders as any[]).forEach((customer: any) => {
    customer.pos?.forEach((po: any) => {
      po.items?.forEach((item: any) => {
        const metadata = { item, customerName: customer.customerName, poNumber: po.poNumber };
        if (item.orderId) orderIdMap.set(item.orderId, metadata);
        const compositeKey = `${item.poItemId}-${item.unitNumber}`;
        compositeKeyMap.set(compositeKey, metadata);
      });
    });
  });
  // Transform selected checkbox keys into ShipmentDialog format
  const items = [];
  selectedPOItems.forEach((key) => {
    const metadata = orderIdMap.get(key) || compositeKeyMap.get(key);
    if (metadata) {
      const { item, customerName, poNumber } = metadata;
      items.push({
        poItemId: item.poItemId,
        orderId: item.orderId || `PO-${item.poItemId}-${item.unitNumber}`,
        quantity: item.quantity ?? 1,
        description: item.description || item.itemName || item.stockModelName || 'Unknown Item',
        customerName,
        poNumber,
      });
    }
  });
  return items.sort(...);
}, [selectedPOItems, poOrders]);

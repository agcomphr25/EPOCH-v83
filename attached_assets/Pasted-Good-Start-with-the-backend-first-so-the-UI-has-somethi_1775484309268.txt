Step 2 — release endpoint

In server/src/routes/manufacturingQueue.ts, add this route before export default router.

router.post("/:id/release", async (req, res) => {
  try {
    const queueId = Number(req.params.id);

    const [queue] = await db
      .select()
      .from(manufacturingQueue)
      .where(eq(manufacturingQueue.id, queueId));

    if (!queue) {
      return res.status(404).json({ message: "Queue item not found" });
    }

    if (queue.queueType !== "KIT") {
      return res.status(400).json({
        message: "Only KIT queue items can be released from this endpoint",
      });
    }

    if (queue.readinessStatus !== "READY") {
      return res.status(400).json({
        message: "Kit is not ready — only READY kits can be released",
      });
    }

    if (
      ["IN_PROGRESS", "COMPLETED", "RELEASED", "CANCELLED"].includes(
        queue.status || ""
      )
    ) {
      return res.status(400).json({
        message: `Cannot release queue item in status ${queue.status}`,
      });
    }

    const [updated] = await db
      .update(manufacturingQueue)
      .set({
        status: "RELEASED",
        releasedAt: new Date(),
        updatedAt: new Date(),
      })
      .where(eq(manufacturingQueue.id, queueId))
      .returning();

    return res.json(updated);
  } catch (error: any) {
    return res.status(500).json({ message: error.message });
  }
});

  const possibleQueues = await db
    .select()
    .from(manufacturingQueue)

        eq(manufacturingQueue.queueType, "KIT")

3. Build Suggestion Aggregator

📁 server/src/lib/quoteSuggestionEngine.ts

export function buildSuggestionSummary(feedbackList) {
  if (!feedbackList.length) return null;

  const avgVariance =
    feedbackList.reduce((sum, f) => sum + Number(f.laborVariancePercent || 0), 0) /
    feedbackList.length;

  const avgActualHours =
    feedbackList.reduce((sum, f) => sum + Number(f.actualLaborHours || 0), 0) /
    feedbackList.length;

  return {
    avgVariancePercent: avgVariance,
    avgActualLaborHours: avgActualHours,
    overrunRisk: avgVariance > 10,
    recommendedAdjustment:
      avgVariance > 0 ? `Increase estimate by ${avgVariance.toFixed(1)}%` : "No increase needed"
  };
}
4. Add Route

📁 server/src/routes/quotes.ts

router.post("/suggestions", async (req, res) => {
  const { partNumber, projectType, customerId } = req.body;

  const feedback = await storage.getQuoteSuggestions({
    partNumber,
    projectType,
    customerId
  });

  const summary = buildSuggestionSummary(feedback);

  res.json({
    matches: feedback,
    summary
  });
});

import { eq } from "drizzle-orm";

async getEstimatingShipping(rfqId: string) {
  return db
    .select()
    .from(estimatingShipping)
    .where(eq(estimatingShipping.rfqId, rfqId));
}

async createEstimatingShipping(data: any) {
  const [row] = await db
    .insert(estimatingShipping)
    .values(data)
    .returning();
  return row;
}

async deleteEstimatingShipping(id: string) {
  await db
    .delete(estimatingShipping)
    .where(eq(estimatingShipping.id, id));
}

  return shippingRows.filter((row) => {

const addShippingRow = () => {
  if (!rfqId) {
    setShippingMessage("Save the RFQ first before adding shipping.");
    return;
  }

  if (shippingScopeFilter === "PART" && !selectedShippingPartId) {
    setShippingMessage("Select an RFQ part first.");
    return;
  }

  setShippingRows((prev) => [
    ...prev,
    {
      ...emptyShippingRow(),
      rfqPartId: shippingScopeFilter === "PART" ? selectedShippingPartId : null,
    },
  ]);
};

const updateShippingRow = (
  index: number,
  field: keyof ShippingRow,
  value: string | number | boolean | null
) => {
  const matchingIndexes = shippingRows
    .map((row, i) => ({ row, i }))
    .filter(({ row }) => {
      if (shippingScopeFilter === "RFQ") return !row.rfqPartId;
      return row.rfqPartId === selectedShippingPartId;
    })
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setShippingRows((prev) => {
    const next = [...prev];
    next[actualIndex] = { ...next[actualIndex], [field]: value };
    return next;
  });
};

const saveShippingRow = async (row: ShippingRow) => {
  if (!rfqId) {
    setShippingMessage("Save the RFQ first before adding shipping.");
    return;
  }

  await apiRequest("POST", `/api/estimating/rfqs/${rfqId}/shipping`, {
    rfqPartId: shippingScopeFilter === "PART" ? row.rfqPartId : null,
    shippingMode: row.shippingMode,
    description: row.description || "",
    method: row.method || "",
    amount: Number(row.amount || 0),
    allocationMethod: row.allocationMethod || "EVEN",
    includeInCustomerPrice: !!row.includeInCustomerPrice,
    notes: row.notes || "",
  });

  setShippingMessage("Shipping row saved.");
  await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-shipping", rfqId] });
};

const deleteShippingRow = async (index: number, row: ShippingRow) => {
  if (row.id) {
    await apiRequest("DELETE", `/api/estimating/shipping/${row.id}`);
    await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-shipping", rfqId] });
    setShippingMessage("Shipping row deleted.");
    return;
  }

  const matchingIndexes = shippingRows
    .map((row, i) => ({ row, i }))
    .filter(({ row }) => {
      if (shippingScopeFilter === "RFQ") return !row.rfqPartId;
      return row.rfqPartId === selectedShippingPartId;
    })
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setShippingRows((prev) => prev.filter((_, i) => i !== actualIndex));
};
Add summary
const shippingSummary = useMemo(() => {
  return filteredShippingRows.reduce(
    (acc, row) => {
      const amount = Number(row.amount || 0);

File:

Add functions following the existing pattern.

Examples:

export async function getP2OpenPOs() {
const result = await pool.query( SELECT COUNT(*) FROM p2_purchase_orders WHERE status NOT IN ('completed','cancelled') );
return Number(result.rows[0].count);
}

export async function getP2PendingBOMs() {
const result = await pool.query( SELECT COUNT(*) FROM p2_purchase_orders WHERE bom_configured = false );
return Number(result.rows[0].count);
}

export async function getP2ItemsInProduction() {
const result = await pool.query( SELECT COUNT(*) FROM p2_serialized_items WHERE status NOT IN ('completed','cancelled') );
return Number(result.rows[0].count);
}

export async function getP2ItemsPendingQC() {
const result = await pool.query( SELECT COUNT(*) FROM p2_serialized_items WHERE status = 'FINAL_QC' );
return Number(result.rows[0].count);
}

export async function getP2ItemsCompletedWeek() {
const result = await pool.query( SELECT COUNT(*) FROM p2_serialized_items WHERE status = 'COMPLETED' AND completed_at > NOW() - INTERVAL '7 days' );
return Number(result.rows[0].count);
}

export async function getCuttingTableActiveItems() {
const result = await pool.query( SELECT COUNT(*) FROM p2_production_orders WHERE status IN ('pending','in_progress') );
return Number(result.rows[0].count);
}

export async function getOpenCreditMemos() {
const result = await pool.query( SELECT COUNT(*) FROM credit_memos WHERE status = 'active' );
return Number(result.rows[0].count);
}

export async function getOpenTickets() {
const result = await pool.query( SELECT COUNT(*) FROM tickets WHERE status NOT IN ('closed','resolved') );
return Number(result.rows[0].count);
}

1. Create Aggregation Service

📁 server/src/lib/commandCenter.ts

export async function getCommandCenterData(storage) {
  const workOrders = await storage.getAllProductionWorkOrders();

  const result = {
    blocked: [],
    atRisk: [],
    ready: [],
    inProgress: [],
    late: []
  };

  for (const wo of workOrders) {
    const readiness = await evaluateWorkOrderReadiness(storage, wo.id);
    const labor = await evaluateWorkOrderLaborStatus(storage, wo.id);

    // BLOCKED
    if (wo.status !== "RELEASED") {
      result.blocked.push({ ...wo, reason: "Not released" });
      continue;
    }

    if (readiness.status !== "READY") {
      result.blocked.push({ ...wo, reason: readiness.reason });
      continue;
    }

    // AT RISK
    if (labor.status === "WARNING") {
      result.atRisk.push({ ...wo, labor });
      continue;
    }

    // IN PROGRESS
    if (wo.status === "IN_PROGRESS") {
      result.inProgress.push({ ...wo, labor });
      continue;
    }

    // READY
    if (wo.status === "RELEASED") {
      result.ready.push(wo);
    }

    // LATE
    if (wo.dueDate && new Date(wo.dueDate) < new Date()) {
      result.late.push(wo);
    }
  }

  return result;
}

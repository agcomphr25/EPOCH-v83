3) Bulk Duplicate Feature ("Order 10 copies")
Backend — modify duplicate endpoint to support count

📄 server/src/routes/orders.ts replace current duplicate route with:

router.post('/duplicate/:orderId', async (req, res) => {
  try {
    const { orderId } = req.params;
    const count = Number(req.body.count ?? 1);

    const existing = await db.select().from(allOrders)
      .where(eq(allOrders.orderId, orderId))
      .limit(1);

    if (!existing.length) return res.status(404).json({ error: 'Original not found' });

    const original = existing[0];
    const results = [];

    for (let i = 0; i < count; i++) {
      const newId = await storage.generateNextOrderId();

      const clone = {
        ...original,
        orderId: newId,
        isReplacement: false,
        replacedOrderId: original.orderId,
        notes: `${original.notes || ''}\n\n🟩 DUPLICATED FROM ${original.orderId}`,
        status: 'DRAFT',
        id: undefined,
        scrappedQuantity: 0,
        departmentHistory: [],
        layupCompletedAt: null,
        pluggingCompletedAt: null,
        cncCompletedAt: null,
        finishCompletedAt: null,
        qcCompletedAt: null,
        shippingCompletedAt: null,
        createdAt: new Date(),
        updatedAt: new Date(),
      };

      const inserted = await storage.createFinalizedOrder(clone);
      results.push(inserted);
    }

    return res.json({
      message: `Duplicated ${count} orders`,
      created: results.map(r => r.orderId),
    });

  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Duplication failed' });
  }
});

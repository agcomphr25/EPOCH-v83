3. Permission Helper Upgrade

📁 permissions.ts

export async function requireScopedCapability(
  storage,
  userId,
  capabilityName,
  context
) {
  const allowed = await storage.userHasScopedCapability(
    userId,
    capabilityName,
    context
  );

  if (!allowed) {
    throw {
      status: 403,
      payload: {
        error: "Forbidden",
        requiredCapability: capabilityName,
        context
      }
    };
  }
}

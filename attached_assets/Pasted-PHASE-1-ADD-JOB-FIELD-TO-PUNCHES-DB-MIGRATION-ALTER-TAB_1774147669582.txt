🔵 PHASE 6 — ADD ACTIVE JOB ENDPOINT
router.get('/active-job', authenticateToken, async (req, res) => {

  const last = await db.query(`
    SELECT job_id, punchType
    FROM punch_events
    WHERE epochEmployeeId = $1
    ORDER BY punchTime DESC
    LIMIT 1
  `, [req.user.id]);

  if (!last[0] || last[0].punchType === 'clock_out') {
    return res.json({ jobId: null });
  }

  res.json({ jobId: last[0].job_id });
});

router.get('/jobs', authenticateToken, async (req, res) => {

  const jobs = await db.query(`
    SELECT id, order_number, department
    FROM production_orders
    WHERE status != 'COMPLETE'
    ORDER BY created_at DESC
    LIMIT 50
  `);

  res.json(jobs);
});

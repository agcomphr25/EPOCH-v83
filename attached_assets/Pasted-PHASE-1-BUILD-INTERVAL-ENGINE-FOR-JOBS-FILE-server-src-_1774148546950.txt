export function buildJobIntervals(punches) {
  const intervals = [];
  let current = null;

  for (const p of punches) {
    if (p.punchType === 'clock_in') {
      current = p;
    }

    if (p.punchType === 'clock_out' && current) {
      intervals.push({
        jobId: current.job_id,
        start: current.punchTime,
        end: p.punchTime,
        hours: (new Date(p.punchTime) - new Date(current.punchTime)) / 3600000
      });
      current = null;
    }
  }

  return intervals;
}

router.get('/admin/job-labor/:jobId', authenticateToken, requireRole('ADMIN'), async (req, res) => {

  const punches = await db.query(`
    SELECT *
    FROM punch_events
    WHERE job_id = $1
    ORDER BY punchTime ASC
  `, [req.params.jobId]);

  const intervals = buildJobIntervals(punches);

  const totalHours = intervals.reduce((sum, i) => sum + i.hours, 0);

  res.json({
    jobId: req.params.jobId,
    totalHours,
    intervals
  });
});
🔵 PHASE 3 — LABOR BY EMPLOYEE (PER JOB)
ADD:

router.get('/admin/job-labor-breakdown/:jobId', authenticateToken, requireRole('ADMIN'), async (req, res) => {

  const punches = await db.query(`
    SELECT epochEmployeeId, punchType, punchTime, job_id
    FROM punch_events
    WHERE job_id = $1
    ORDER BY punchTime ASC
  `, [req.params.jobId]);

  const byEmployee = {};

  for (const p of punches) {
    if (!byEmployee[p.epochEmployeeId]) {
      byEmployee[p.epochEmployeeId] = [];
    }
    byEmployee[p.epochEmployeeId].push(p);
  }

  const result = [];

  for (const empId in byEmployee) {
    const intervals = buildJobIntervals(byEmployee[empId]);

    const hours = intervals.reduce((sum, i) => sum + i.hours, 0);

    result.push({
      employeeId: empId,
      hours
    });
  }

  res.json(result);
});

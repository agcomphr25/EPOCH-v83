🧩 2. CREATE READINESS ENGINE

📁 server/src/lib/workOrderReadiness.ts

export async function evaluateWorkOrderReadiness(storage, workOrderId) {
  const travelers = await storage.getTravelersByProductionWorkOrderId(workOrderId);

  // Check if travelers exist
  if (!travelers.length) {
    return { status: "BLOCKED", reason: "No travelers created" };
  }

  // Check materials
  const materialsReady = await storage.checkWorkOrderMaterialAvailability(workOrderId);
  if (!materialsReady) {
    return { status: "PARTIAL", reason: "Materials not fully allocated" };
  }

  // Check training readiness (basic)
  const trainingReady = await storage.checkWorkOrderTrainingCoverage(workOrderId);
  if (!trainingReady) {
    return { status: "PARTIAL", reason: "Training gaps exist" };
  }

  return { status: "READY" };
}

🌐 4. RELEASE ROUTE

📁 server/src/routes/workOrders.ts

Add:
router.post("/:id/release", async (req, res) => {
  const readiness = await evaluateWorkOrderReadiness(storage, req.params.id);

  if (readiness.status !== "READY") {
    return res.status(400).json({
      error: "Work order not ready for release",
      details: readiness
    });
  }

  const updated = await storage.updateWorkOrderStatus(
    req.params.id,
    "RELEASED"
  );

  res.json(updated);
});

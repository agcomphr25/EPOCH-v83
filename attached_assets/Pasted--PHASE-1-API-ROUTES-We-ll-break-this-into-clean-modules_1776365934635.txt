🧱 1. CHARGE CODES ROUTES
📁 server/routes/chargeCodes.ts
import { Router } from "express";
import { z } from "zod";
import { storage } from "../storage";

const router = Router();

const createSchema = z.object({
  code: z.string(),
  name: z.string(),
  description: z.string().optional(),
  chargeType: z.enum(["DIRECT","INDIRECT","UNALLOWABLE"]),
  projectId: z.string().optional(),
  workOrderId: z.string().optional(),
  department: z.string().optional(),
});

router.post("/", async (req, res) => {
  const data = createSchema.parse(req.body);

  const cc = await storage.createChargeCode(data);
  res.json(cc);
});

router.get("/", async (_req, res) => {
  const list = await storage.getChargeCodes();
  res.json(list);
});

export default router;

🧱 5. WORK SESSIONS (CORE ENGINE)
📁 server/routes/workSessions.ts
🔹 START SESSION (AUTO CHARGE CODE + CHECK HOURS)
router.post("/start", async (req, res) => {
  const { employeeId, travelerId, workOrderId } = req.body;

  // 🔥 Resolve context
  const context = await storage.resolveWorkContext({
    travelerId,
    workOrderId,
  });

  const chargeCodeId = context.chargeCodeId;

  // 🔥 Check labor authorization
  const auth = await storage.getLaborAuthorization(chargeCodeId, context);

  const remaining =
    auth.authorizedHours +
    auth.approvedExtraHours -
    auth.consumedHours;

  if (remaining <= 0 && auth.hardStop) {
    return res.status(400).json({
      error: "NO_HOURS_REMAINING",
      message: "Request additional hours",
    });
  }

  const session = await storage.startWorkSession({
    employeeId,
    ...context,
    chargeCodeId,
    startedAt: new Date(),
  });

  res.json(session);
});
🔹 STOP SESSION (UPDATE HOURS)
router.post("/:id/stop", async (req, res) => {
  const session = await storage.getSession(req.params.id);

  const endedAt = new Date();

  const hours =
    (endedAt.getTime() - session.startedAt.getTime()) / 3600000;

  await storage.stopSession(req.params.id, endedAt, hours);

  // 🔥 Update consumed hours
  await storage.incrementConsumedHours(
    session.chargeCodeId,
    session.projectId,
    hours
  );

  res.json({ success: true });
});

STEP 4 — Implement unified filtering

  search: '',

Implement a filter function:

function applyFilters(requests, filters) {
  return requests.filter(r => {

    if (filters.status && r.status !== filters.status)
      return false

    if (filters.search) {
      const s = filters.search.toLowerCase()

      const partNumber = (r.partNumber ?? '').toLowerCase()
      const partName = (r.partName ?? '').toLowerCase()
      const vendorName = (r.vendorName ?? '').toLowerCase()

      if (
        !partNumber.includes(s) &&
        !partName.includes(s) &&
        !vendorName.includes(s)
      ) return false
    }

    return true
  })
}

STEP 5 — Create filtered dataset

const filteredRequests = useMemo(

STEP 6 — Create vendor grouping
--------------------------------

function groupByVendor(requests) {
  const map = {}

  for (const r of requests) {

    const key =
      r.orderMethod === 'WEBSITE'
        ? 'Website Orders'
        : r.vendorName ?? 'Unassigned'

    if (!map[key]) {
      map[key] = {
        vendor: key,
        vendorId: r.vendorId,
        requests: []
      }
    }

    map[key].requests.push(r)
  }

  return Object.values(map)
}

const vendorGroups = useMemo(
  () => groupByVendor(filteredRequests),
  [filteredRequests]
)

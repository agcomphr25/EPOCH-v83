🧮 3. LABOR STATUS HELPER (NEW FILE)

📁 server/src/lib/laborStatus.ts

export async function evaluateWorkOrderLaborStatus(storage, workOrderId, department?) {
  const workOrder = await storage.getWorkOrderById(workOrderId);

  const totalHours = await storage.getLaborHoursByWorkOrder(workOrderId);

  let departmentHours = 0;
  let departmentBudget = null;

  if (department && workOrder.departmentBudgets) {
    departmentHours = await storage.getLaborHoursByWorkOrderAndDepartment(workOrderId, department);
    departmentBudget = workOrder.departmentBudgets?.[department] || null;
  }

  const totalBudget = Number(workOrder.totalBudgetHours || 0);

  const percentUsed = totalBudget > 0 ? (totalHours / totalBudget) * 100 : 0;

  let status = "OK";
  if (percentUsed >= 100) status = "BLOCKED";
  else if (percentUsed >= 80) status = "WARNING";

  return {
    totalHours,
    departmentHours,
    totalBudget,
    departmentBudget,
    percentUsed,
    status
  };
}

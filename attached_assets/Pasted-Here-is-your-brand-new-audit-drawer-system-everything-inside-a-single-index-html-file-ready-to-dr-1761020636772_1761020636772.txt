<script>
const state = {
  open:false, page:0, pageSize:5, filter:"all",
  entityType:"WorkOrder", entityId:"WO-000123",
};

let MOCK_EVENTS = (() => {
  const now = Date.now();
  return [
    {
      id: cid(), ts: new Date(now - 4000000).toISOString(), actorName: "Jenny Jarrett", action: "create",
      reason: "Initial creation", entityType:"WorkOrder", entityId:"WO-000123",
      fieldsChanged: { before:null, after:{ status:"Open", area:"Layup", qty:4 } }
    },
    {
      id: cid(), ts: new Date(now - 3500000).toISOString(), actorName: "Kara Ko", action: "update",
      reason: "Update qty", entityType:"WorkOrder", entityId:"WO-000123",
      fieldsChanged: { before:{ qty:4 }, after:{ qty:6 } }
    },
    {
      id: cid(), ts: new Date(now - 3000000).toISOString(), actorName: "AutoBot", action: "approve",
      reason: "Auto-approved by rule", entityType:"WorkOrder", entityId:"WO-000123"
    },
    {
      id: cid(), ts: new Date(now - 2000000).toISOString(), actorName: "Steve Nash", action: "progress",
      reason: "Moved to Plugging", entityType:"WorkOrder", entityId:"WO-000123",
      meta:{ fromArea:"Layup", toArea:"Plugging" }
    }
  ];
})();

const drawer = document.getElementById('drawer');
const backdrop = document.getElementById('backdrop');
const openBtn = document.getElementById('openBtn');
const closeBtn = document.getElementById('closeBtn');
const eventsEl = document.getElementById('events');
const titleMeta = document.getElementById('titleMeta');
const countMeta = document.getElementById('countMeta');
const exportLink = document.getElementById('exportLink');
const tabs = document.querySelectorAll('.tab');
const prevBtn = document.getElementById('prev');
const nextBtn = document.getElementById('next');
const progressBtn = document.getElementById('progressBtn');
const areaEl = document.getElementById('area');
const lane = document.getElementById('lane');

openBtn.onclick = () => {
  state.entityType = document.getElementById('entityType').value;
  state.entityId = document.getElementById('entityId').value;
  openDrawer();
};

closeBtn.onclick = closeDrawer;
backdrop.onclick = closeDrawer;
tabs.forEach(tab => tab.onclick = () => {
  tabs.forEach(t => t.classList.remove('active'));
  tab.classList.add('active');
  state.filter = tab.dataset.filter;
  state.page = 0;
  render();
});
prevBtn.onclick = () => { if(state.page > 0) { state.page--; render(); } };
nextBtn.onclick = () => {
  const total = getFiltered().length;
  if((state.page+1)*state.pageSize < total) { state.page++; render(); }
};

progressBtn.onclick = () => {
  const from = areaEl.textContent.trim();
  const to = from === "Layup" ? "Plugging" : from === "Plugging" ? "CNC" : "Final";
  areaEl.textContent = to;
  MOCK_EVENTS.unshift({
    id: cid(), ts: new Date().toISOString(), actorName: "Glenn G", action: "progress",
    reason: `Moved to ${to}`, entityType: state.entityType, entityId: state.entityId,
    meta: { fromArea: from, toArea: to }
  });
  render();
};

function openDrawer() {
  state.open = true;
  drawer.classList.add('open');
  backdrop.classList.add('open');
  render();
}
function closeDrawer() {
  state.open = false;
  drawer.classList.remove('open');
  backdrop.classList.remove('open');
}
function getFiltered() {
  const all = MOCK_EVENTS.filter(e => e.entityType === state.entityType && e.entityId === state.entityId);
  if (state.filter === "changes") return all.filter(e => e.fieldsChanged);
  if (state.filter === "progress") return all.filter(e => e.action === "progress");
  if (state.filter === "sign") return all.filter(e => ["sign","approve"].includes(e.action));
  return all;
}
function renderLane(events){
  const progs = events.filter(e => e.action === "progress" && e.meta?.toArea);
  lane.innerHTML = progs.length === 0 ? "<em>No progress yet</em>" :
    progs.sort((a,b)=>new Date(a.ts)-new Date(b.ts))
         .map(p => `<span class="lane-chip">${p.meta.toArea} · <span class="ev-meta">${new Date(p.ts).toLocaleString()}</span></span>`)
         .join("");
}

function render() {
  const all = getFiltered();
  const page = all.slice(state.page * state.pageSize, (state.page+1)*state.pageSize);
  titleMeta.textContent = `${state.entityType} • ${state.entityId}`;
  countMeta.textContent = `${all.length} events`;
  renderLane(all);
  eventsEl.innerHTML = page.map(renderEvent).join('') || "<div style='padding:1em'>No events.</div>";
  const blob = new Blob([JSON.stringify(all,null,2)], { type:"application/json" });
  exportLink.href = URL.createObjectURL(blob);
  exportLink.download = `${state.entityType}-${state.entityId}-audit.json`;
  prevBtn.disabled = state.page === 0;
  nextBtn.disabled = (state.page+1)*state.pageSize >= all.length;
}

function renderEvent(ev) {
  const ts = new Date(ev.ts).toLocaleString();
  const reason = ev.reason ? `<div><strong>Reason:</strong> ${escape(ev.reason)}</div>` : "";
  const changes = ev.fieldsChanged ? renderDiff(ev.fieldsChanged) : "";
  return `
    <div class="event">
      <div class="ev-head">
        <span><strong>${escape(ev.actorName)}</strong> <span class="ev-meta">· ${ts}</span></span>
        <span class="chip">${escape(ev.action)}</span>
      </div>
      ${reason}
      ${changes}
    </div>
  `;
}
function renderDiff(ch) {
  return `<div class="diff">
    ${Object.keys(ch.after || {}).map(k => `
      <div class="diff-row">
        <div>${escape(k)}</div>
        <div class="before">${escape((ch.before||{})[k])}</div>
        <div class="after"><strong>${escape((ch.after||{})[k])}</strong></div>
      </div>`).join('')}
  </div>`;
}
function cid() {
  return "evt_" + Math.random().toString(36).slice(2) + "_" + Date.now().toString(36);
}
function escape(s) {
  return String(s ?? "").replace(/[&<>"']/g, c => ({
    '&':"&amp;", '<':"&lt;", '>':"&gt;", '"':"&quot;", "'":"&#39;"
  }[c]));
}

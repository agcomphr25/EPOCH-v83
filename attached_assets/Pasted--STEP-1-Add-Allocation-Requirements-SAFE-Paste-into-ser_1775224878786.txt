🔧 STEP 3 — Readiness Engine (NEW)
📍 Create file:

server/src/services/queueReadinessService.ts

import { db } from "../db";
import { manufacturingQueue, allocationRequirements } from "../../schema";
import { eq } from "drizzle-orm";

export async function evaluateQueueReadiness(queueId: number) {
  const requirements = await db
    .select()
    .from(allocationRequirements)
    .where(eq(allocationRequirements.manufacturingQueueId, queueId));

  let total = 0;
  let ready = 0;

  const blocking: string[] = [];

  for (const req of requirements) {
    if (!req.isCritical) continue;

    total++;

    const required = Number(req.requiredQty || 0);
    const allocated = Number(req.allocatedQty || 0);

    if (allocated >= required) {
      ready++;
    } else {
      blocking.push(
        `${req.requiredPartNumber} short (${allocated}/${required})`
      );
    }
  }

  let readiness = "READY";
  let status = "PENDING";

  if (blocking.length > 0) {
    status = "BLOCKED";
    readiness = ready === 0 ? "NOT_READY" : "PARTIAL";
  }

  const percent = total === 0 ? 100 : Math.round((ready / total) * 100);

  await db
    .update(manufacturingQueue)
    .set({
      readinessStatus: readiness,
      status,
      percentReady: percent.toString(),
      blockedReason: blocking.join(", "),
      updatedAt: new Date(),
    })
    .where(eq(manufacturingQueue.id, queueId));

  return { status, readiness, percent, blocking };
}

Add helper:
export function mapQueueType(category: string) {
  switch (category) {
    case "PACKET":
      return { queueType: "LAYUP", department: "Cutting Table" };

    case "CORE":
      return { queueType: "CORE", department: "Cores" };

    case "KIT":
      return { queueType: "KIT", department: "Kitting" };

    case "SUB_ASSEMBLY":
      return { queueType: "SUB_ASSEMBLY", department: "Assembly" };

    case "ASSEMBLY":
      return { queueType: "ASSEMBLY", department: "Assembly" };

    default:
      return { queueType: "LAYUP", department: "Cutting Table" };
  }
}

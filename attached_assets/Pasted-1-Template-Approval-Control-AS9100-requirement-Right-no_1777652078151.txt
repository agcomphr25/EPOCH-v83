Backend Execution Pipeline
async function createProductionControls(wadId, selection) {

  if (selection.routingRequired) {
    const routing = await createRoutingFromTemplate(selection.routingTemplateId)
  }

  if (selection.travelerRequired) {
    const traveler = await createTravelerFromRouting(routing.id)
  }

  if (selection.inProcessInspectionRequired) {
    await attachQCFromTemplate(selection.qcTemplateId, traveler.id)
  }

  if (selection.workInstructionRequired) {
    await linkWorkInstructions(wadId, selection.workInstructionTemplateId)
  }

  if (selection.specSheetRequired) {
    await linkSpecSheet(wadId, selection.specSheetTemplateId)
  }

  if (selection.certRequired) {
    await addCertificationRequirement(traveler.id)
  }

  return provisionSummary
}

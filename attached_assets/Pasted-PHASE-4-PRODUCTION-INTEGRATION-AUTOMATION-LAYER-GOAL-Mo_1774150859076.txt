🔵 PHASE 2 — ADD “ACTIVE JOB” CONTEXT
FILE:
server/src/routes/timekeeping.ts

---

ADD:

router.get('/active-context', authenticateToken, async (req, res) => {

  const last = await db.query(`
    SELECT job_id, punchType
    FROM punch_events
    WHERE epochEmployeeId = $1
    ORDER BY punchTime DESC
    LIMIT 1
  `, [req.user.id]);

  if (!last[0] || last[0].punchType === 'clock_out') {
    return res.json({ activeJobId: null });
  }

  res.json({
    activeJobId: last[0].job_id
  });
});

🔴 PHASE 4 — CREATE NATIVE CLOCK ROUTE
Create new native punch route.

FILE:
server/src/routes/timekeeping.ts

---

IMPLEMENT:

import { Router } from 'express';
import { randomUUID } from 'crypto';

const router = Router();

router.post('/punch', authenticateToken, async (req, res) => {
  try {
    const { type } = req.body; // 'clock_in' | 'clock_out'

    if (!type) {
      return res.status(400).json({ error: 'type required' });
    }

    const punchType = type === 'clock_in' ? 'clock_in' : 'clock_out';

    await db.query(`
      INSERT INTO punch_events (
        id,
        externalPunchId,
        canonicalId,
        epochEmployeeId,
        punchType,
        punchTime,
        source
      )
      VALUES ($1,$2,$3,$4,$5,NOW(),'epoch_native')
    `, [
      randomUUID(),
      randomUUID(),
      req.user.canonicalId,
      req.user.id,
      punchType
    ]);

    res.json({ success: true });

  } catch (e) {
    console.error(e);
    res.status(500).json({ error: 'failed to punch' });
  }
});

export default router;

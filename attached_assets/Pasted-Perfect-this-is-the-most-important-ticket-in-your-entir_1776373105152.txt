🧠 1. CREATE GATE ENGINE

📁 server/src/lib/travelerGates.ts

export async function evaluateTravelerStartGates(storage, travelerId, employeeId, stepId) {
  const step = await storage.getTravelerStepById(stepId);

  // 1. Sequence Gate
  const prevStep = await storage.getPreviousStep(step);
  if (prevStep && prevStep.status !== "COMPLETE") {
    return { allowed: false, reason: "Previous step not complete" };
  }

  // 2. Training Gate
  const authorized = await storage.checkTravelerAuthorization(travelerId, employeeId);
  if (!authorized) {
    return { allowed: false, reason: "Training certification required" };
  }

  // 3. Material Gate
  const materialsOk = await storage.checkMaterialTraceability(travelerId, step.department);
  if (!materialsOk) {
    return { allowed: false, reason: "Material not allocated or traceable" };
  }

  return { allowed: true };
}
🧠 Finish Gate
export async function evaluateTravelerFinishGates(storage, travelerId, stepId) {
  const tasks = await storage.getTasksByStep(stepId);

  // QC Gate
  const incompleteTasks = tasks.filter(t => t.required && !t.completed);
  if (incompleteTasks.length > 0) {
    return { allowed: false, reason: "Required QC tasks incomplete" };
  }

  return { allowed: true };
}

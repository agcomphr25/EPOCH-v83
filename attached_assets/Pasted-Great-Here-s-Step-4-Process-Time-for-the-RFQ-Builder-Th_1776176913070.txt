Add:

import { estimatingProcessRows } from "../schema"; // adjust path
import { eq } from "drizzle-orm";

async getEstimatingProcessRows(rfqId: string) {
  return db
    .select()
    .from(estimatingProcessRows)
    .where(eq(estimatingProcessRows.rfqId, rfqId));
}

async createEstimatingProcessRow(data: any) {
  const [row] = await db
    .insert(estimatingProcessRows)
    .values(data)
    .returning();
  return row;
}

async deleteEstimatingProcessRow(id: string) {
  await db
    .delete(estimatingProcessRows)
    .where(eq(estimatingProcessRows.id, id));
}

Add helpers:

const filteredProcessRows = useMemo(() => {
  if (!selectedProcessPartId) return [];
  return processRows.filter((row) => row.rfqPartId === selectedProcessPartId);
}, [processRows, selectedProcessPartId]);

const addProcessRow = () => {
  if (!rfqId) {
    setProcessMessage("Save the RFQ first before adding process rows.");
    return;
  }

  if (!selectedProcessPartId) {
    setProcessMessage("Select an RFQ part first.");
    return;
  }

  setProcessRows((prev) => [...prev, emptyProcessRow(selectedProcessPartId)]);
};

const updateProcessRow = (
  index: number,
  field: keyof ProcessRow,
  value: string | number
) => {
  const matchingIndexes = processRows
    .map((row, i) => ({ row, i }))
    .filter(({ row }) => row.rfqPartId === selectedProcessPartId)
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setProcessRows((prev) => {
    const next = [...prev];
    next[actualIndex] = { ...next[actualIndex], [field]: value };
    return next;
  });
};

const saveProcessRow = async (row: ProcessRow) => {
  if (!rfqId) {
    setProcessMessage("Save the RFQ first before adding process rows.");
    return;
  }

  if (!row.rfqPartId) {
    setProcessMessage("A process row must be tied to an RFQ part.");
    return;
  }

  if (!row.departmentName.trim()) {
    setProcessMessage("Department name is required.");
    return;
  }

  await apiRequest("POST", `/api/estimating/rfqs/${rfqId}/process-rows`, {
    rfqPartId: row.rfqPartId,
    departmentName: row.departmentName,
    sourceType: row.sourceType,
    setupHours: Number(row.setupHours || 0),
    hoursPerPart: Number(row.hoursPerPart || 0),
    hourlyRate: Number(row.hourlyRate || 0),
    notes: row.notes || "",
  });

  setProcessMessage("Process row saved.");
  await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-process-rows", rfqId] });
};

const deleteProcessRow = async (index: number, row: ProcessRow) => {
  if (row.id) {
    await apiRequest("DELETE", `/api/estimating/process-rows/${row.id}`);
    await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-process-rows", rfqId] });
    setProcessMessage("Process row deleted.");
    return;
  }

  const matchingIndexes = processRows
    .map((row, i) => ({ row, i }))
    .filter(({ row }) => row.rfqPartId === selectedProcessPartId)
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setProcessRows((prev) => prev.filter((_, i) => i !== actualIndex));
};

Add summary calculations:

const processSummary = useMemo(() => {
  return filteredProcessRows.reduce(
    (acc, row) => {
      const setupHours = Number(row.setupHours || 0);
      const hoursPerPart = Number(row.hoursPerPart || 0);
      const hourlyRate = Number(row.hourlyRate || 0);

      const setupCost = setupHours * hourlyRate;
      const recurringCostPerPart = hoursPerPart * hourlyRate;

      acc.setupCost += setupCost;
      acc.recurringCostPerPart += recurringCostPerPart;
      acc.totalPerPart += recurringCostPerPart;

      return acc;
    },
    {
      setupCost: 0,
      recurringCostPerPart: 0,
      totalPerPart: 0,
    }
  );
}, [filteredProcessRows]);

const selectedProcessPart = useMemo(
  () => parts.find((p) => p.id === selectedProcessPartId),
  [parts, selectedProcessPartId]
);

const selectedProcessQty = Number(selectedProcessPart?.quantity ?? 0);

const processExtendedTotal =
  processSummary.setupCost + processSummary.recurringCostPerPart * selectedProcessQty;

export function pairPunches(punches) {
  const intervals = [];
  let currentIn = null;

  for (const p of punches) {
    if (p.punchType === 'clock_in') {
      currentIn = p;
    }

    if (p.punchType === 'clock_out' && currentIn) {
      intervals.push({
        clockIn: currentIn.punchTime,
        clockOut: p.punchTime,
        durationHours: (new Date(p.punchTime) - new Date(currentIn.punchTime)) / 3600000
      });
      currentIn = null;
    }
  }

  return intervals;
}

router.get('/hours', authenticateToken, async (req, res) => {
  const punches = await db.query(`
    SELECT punchType, punchTime
    FROM punch_events
    WHERE epochEmployeeId = $1
    ORDER BY punchTime ASC
  `, [req.user.id]);

  const intervals = pairPunches(punches);

  const totalHours = intervals.reduce((sum, i) => sum + i.durationHours, 0);

  res.json({
    intervals,
    totalHours
  });
});

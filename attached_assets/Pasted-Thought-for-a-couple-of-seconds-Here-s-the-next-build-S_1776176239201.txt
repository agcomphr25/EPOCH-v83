import { eq } from "drizzle-orm";

async getEstimatingBomLines(rfqId: string) {
  return db
    .select()
    .from(estimatingBomLines)
    .where(eq(estimatingBomLines.rfqId, rfqId));
}

async createEstimatingBomLine(data: any) {
  const [row] = await db
    .insert(estimatingBomLines)
    .values(data)
    .returning();
  return row;
}

async deleteEstimatingBomLine(id: string) {
  await db
    .delete(estimatingBomLines)
    .where(eq(estimatingBomLines.id, id));
}

const filteredBomLines = useMemo(() => {
  if (!selectedBomPartId) return [];

const addBomLine = () => {
  if (!rfqId) {
    setBomMessage("Save the RFQ first before adding BOM lines.");
    return;
  }

  if (!selectedBomPartId) {
    setBomMessage("Select an RFQ part first.");
    return;
  }

  setBomLines((prev) => [...prev, emptyBomLine(selectedBomPartId)]);
};

const updateBomLine = (
  index: number,
  field: keyof BomLineRow,
  value: string | number | boolean | null
) => {
  const matchingIndexes = bomLines
    .map((line, i) => ({ line, i }))
    .filter(({ line }) => line.rfqPartId === selectedBomPartId)
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setBomLines((prev) => {
    const next = [...prev];
    next[actualIndex] = { ...next[actualIndex], [field]: value };
    return next;
  });
};

const saveBomLine = async (row: BomLineRow) => {
  if (!rfqId) {
    setBomMessage("Save the RFQ first before adding BOM lines.");
    return;
  }

  if (!row.rfqPartId) {
    setBomMessage("A BOM line must be tied to an RFQ part.");
    return;
  }

  if (!row.description.trim()) {
    setBomMessage("BOM line description is required.");
    return;
  }

  await apiRequest("POST", `/api/estimating/rfqs/${rfqId}/bom-lines`, {
    rfqPartId: row.rfqPartId,
    inventoryItemId: row.inventoryItemId ?? null,
    childPartAgNumber: row.childPartAgNumber || "",
    description: row.description,
    category: row.category,
    quantityPerPart: Number(row.quantityPerPart || 0),
    uom: row.uom || "EA",
    estimatedUnitCost: Number(row.estimatedUnitCost || 0),
    scrapPercent: Number(row.scrapPercent || 0),
    isEstimated: !!row.isEstimated,
    isDraftInventoryItem: !!row.isDraftInventoryItem,
    vendorNameSnapshot: row.vendorNameSnapshot || "",
    materialSpec: row.materialSpec || "",
    notes: row.notes || "",
  });

  setBomMessage("BOM line saved.");
  await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-bom-lines", rfqId] });
};

const deleteBomLine = async (index: number, row: BomLineRow) => {
  if (row.id) {
    await apiRequest("DELETE", `/api/estimating/bom-lines/${row.id}`);
    await queryClient.invalidateQueries({ queryKey: ["estimating-rfq-bom-lines", rfqId] });
    setBomMessage("BOM line deleted.");
    return;
  }

  const matchingIndexes = bomLines
    .map((line, i) => ({ line, i }))
    .filter(({ line }) => line.rfqPartId === selectedBomPartId)
    .map(({ i }) => i);

  const actualIndex = matchingIndexes[index];
  if (actualIndex === undefined) return;

  setBomLines((prev) => prev.filter((_, i) => i !== actualIndex));
};

const bomSummary = useMemo(() => {
  return filteredBomLines.reduce(
    (acc, row) => {
      const qty = Number(row.quantityPerPart || 0);
      const unit = Number(row.estimatedUnitCost || 0);
      const scrap = Number(row.scrapPercent || 0) / 100;
      const costPerPart = qty * unit * (1 + scrap);

      acc.totalPerPart += costPerPart;

      if (row.category === "PREPREG") acc.prepreg += costPerPart;
      if (row.category === "CONSUMABLE") acc.consumables += costPerPart;
      if (row.category === "PAINT") acc.paint += costPerPart;
      if (row.category === "HARDWARE") acc.hardware += costPerPart;
      if (row.category === "PACKAGING") acc.packaging += costPerPart;
      if (row.category === "ADHESIVE") acc.adhesive += costPerPart;
      if (row.category === "OTHER") acc.other += costPerPart;

      return acc;
    },
    {
      totalPerPart: 0,
      prepreg: 0,
      consumables: 0,
      paint: 0,
      hardware: 0,
      packaging: 0,
      adhesive: 0,
      other: 0,
    }
  );
}, [filteredBomLines]);

const selectedBomPart = useMemo(
  () => parts.find((p) => p.id === selectedBomPartId),
  [parts, selectedBomPartId]
);

const selectedBomPartQty = Number(selectedBomPart?.quantity ?? 0);
const bomExtendedTotal = bomSummary.totalPerPart * selectedBomPartQty;

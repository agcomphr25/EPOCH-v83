🧩 2. REPLACE RESERVATION FULFILLMENT (CRITICAL FIX)

Replace this:

await storage.fulfillLotReservation(match.id);

With this:

if (match) {
  const reservedQty = parseFloat(String(match.quantityReserved));

  if (consumed >= reservedQty) {
    // Full fulfill
    await storage.fulfillLotReservation(match.id);
  } else {
    // Partial fulfill → reduce reservation
    const newReserved = reservedQty - consumed;

    await db
      .update(materialLotReservations)
      .set({ quantityReserved: newReserved.toString() })
      .where(eq(materialLotReservations.id, match.id));
  }
}

👉 Now reservations stay accurate.

Add:

// Find inventory balance for this material
const balance = await db.query.inventoryBalances.findFirst({
  where: (b, { eq }) => eq(b.agPartNumber, lot.materialPartNumber),
});

if (balance) {
  const newOnHand = balance.quantityOnHand - consumed;

  let newAllocated = balance.quantityAllocated;
  if (match) {
    newAllocated = Math.max(0, newAllocated - consumed);
  }

  const newAvailable = newOnHand - newAllocated;

  await storage.updateInventoryBalance(balance.id, {
    quantityOnHand: newOnHand,
    quantityAllocated: newAllocated,
    quantityAvailable: newAvailable,
  });
}

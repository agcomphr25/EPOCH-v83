2. Add Validation Helper

📁 server/src/lib/projectClosingValidation.ts

export function validateProjectClosing(closing) {
  const missing = [];

  if (!closing.summary) missing.push("summary");
  if (!closing.whatWentWrong) missing.push("whatWentWrong");
  if (!closing.strengths) missing.push("strengths");
  if (!closing.opportunities) missing.push("opportunities");
  if (!closing.nextProjectRecommendations) missing.push("nextProjectRecommendations");

  return {
    valid: missing.length === 0,
    missing
  };
}
3. Hook Into Project Status Change

📁 server/src/routes/projects.ts

Find where project status is updated.

BEFORE allowing:
status = "COMPLETED" or "CLOSED"
ADD:
const closing = await storage.getProjectClosingByProjectId(projectId);

if (!closing) {
  return res.status(400).json({
    error: "Project cannot be completed without closing record"
  });
}

const validation = validateProjectClosing(closing);

if (!validation.valid) {
  return res.status(400).json({
    error: "Project closing incomplete",
    missingFields: validation.missing
  });
}

5. Add Approval Endpoint

📁 server/src/routes/projectClosings.ts

router.post("/:projectId/closing/approve", async (req, res) => {
  const { approvedBy } = req.body;

  const closing = await storage.getProjectClosingByProjectId(req.params.projectId);

  if (!closing) {
    return res.status(404).json({ error: "Closing not found" });
  }

  const updated = await storage.updateProjectClosing(closing.id, {
    approvedBy,
    approvedAt: new Date()
  });

  res.json(updated);
});

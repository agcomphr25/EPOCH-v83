export function getPayPeriod(date = new Date()) {
  const d = new Date(date);
  const year = d.getFullYear();
  const month = d.getMonth();

  const firstHalfStart = new Date(year, month, 1);
  const firstHalfEnd = new Date(year, month, 15, 23, 59, 59);

  const secondHalfStart = new Date(year, month, 16);
  const secondHalfEnd = new Date(year, month + 1, 0, 23, 59, 59);

  if (d.getDate() <= 15) {
    return { start: firstHalfStart, end: firstHalfEnd };
  }

  return { start: secondHalfStart, end: secondHalfEnd };
}

import { Router } from 'express';
import { storage } from '../storage';

const router = Router();

// Get all accessories
router.get('/', async (req, res) => {
  const data = await storage.getAllMetalAccessories();
  res.json(data);
});

// Get demand summary
router.get('/demands', async (req, res) => {
  const data = await storage.getMetalAccessoriesDemands();
  res.json(data);
});

// Bottom metal demand summary
router.get('/bottom-metal-demands', async (req, res) => {
  const data = await storage.getBottomMetalDemandsSummary();
  res.json(data);
});

// Bottom metal demand full list
router.get('/bottom-metal-demands/all', async (req, res) => {
  const data = await storage.getBottomMetalDemands();
  res.json(data);
});

// CRUD
router.post('/', async (req, res) => {
  const created = await storage.createMetalAccessory(req.body);
  res.json(created);
});

router.put('/:id', async (req, res) => {
  const updated = await storage.updateMetalAccessory(
    Number(req.params.id),
    req.body
  );
  res.json(updated);
});

router.delete('/:id', async (req, res) => {
  await storage.deleteMetalAccessory(Number(req.params.id));
  res.json({ success: true });
});

export default router;

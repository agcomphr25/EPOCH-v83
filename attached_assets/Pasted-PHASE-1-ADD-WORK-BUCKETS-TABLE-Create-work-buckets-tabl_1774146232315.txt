CREATE TABLE work_buckets (

  name text NOT NULL,
  type text NOT NULL, -- DIRECT, INDIRECT, NON_WORK
  active boolean DEFAULT true,
  created_at timestamp DEFAULT now()
);

router.get('/buckets', authenticateToken, async (req, res) => {
  const buckets = await db.query(`
    SELECT id, name, type
    FROM work_buckets
    WHERE active = true
    ORDER BY name
  `);
  res.json(buckets);
});

🔵 PHASE 10 — BASIC JOB COST VIEW (FOUNDATION)
Add aggregation endpoint.

router.get('/admin/labor-by-bucket', authenticateToken, requireRole('ADMIN'), async (req, res) => {

  const rows = await db.query(`
    SELECT
      work_bucket_id,
      SUM(
        CASE
          WHEN punchType = 'clock_out'
          THEN 1
          ELSE 0
        END
      ) as punches
    FROM punch_events
    GROUP BY work_bucket_id
  `);

  res.json(rows);
});

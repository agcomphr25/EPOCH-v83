export const productionWorkOrders = pgTable("production_work_orders", {
  id: uuid("id").defaultRandom().primaryKey(),

  workOrderNumber: text("work_order_number").notNull().unique(),

  projectId: uuid("project_id")
    .notNull()
    .references(() => projects.id, { onDelete: "cascade" }),

  partNumber: text("part_number").notNull(),
  description: text("description"),

  quantity: integer("quantity").default(1),

  status: text("status").default("PLANNED"), 
  // PLANNED | READY | RELEASED | IN_PROGRESS | COMPLETE | CLOSED

  departmentBudgets: jsonb("department_budgets").default({}),
  totalBudgetHours: numeric("total_budget_hours"),

  startDate: timestamp("start_date"),
  dueDate: timestamp("due_date"),

  createdAt: timestamp("created_at").defaultNow(),
  updatedAt: timestamp("updated_at").defaultNow()
});

🧩 4. STORAGE LAYER (server/storage.ts)

Add these methods:

async createProductionWorkOrder(data) {
  return await db.insert(productionWorkOrders).values(data).returning();
}

async getWorkOrdersByProject(projectId: string) {
  return await db
    .select()
    .from(productionWorkOrders)
    .where(eq(productionWorkOrders.projectId, projectId));
}

async getWorkOrderById(id: string) {
  return await db
    .select()
    .from(productionWorkOrders)
    .where(eq(productionWorkOrders.id, id))
    .limit(1);
}

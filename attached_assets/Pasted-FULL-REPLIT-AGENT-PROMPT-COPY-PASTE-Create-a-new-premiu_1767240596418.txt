JAVASCRIPT SCROLL ANIMATION (scrollDepth.js):

const cards = document.querySelectorAll('.depth-card');

function animateDepth(){
const vh = window.innerHeight;
cards.forEach(card=>{
const rect=card.getBoundingClientRect();
const dist=Math.abs(rect.top+rect.height/2 - vh/2);
const norm=Math.min(dist/(vh/2),1);
const scale=1.03 - norm*.03;
const shadow=0 ${25 - norm*20}px ${50 - norm*35}px rgba(0,0,0,${.55 - norm*.35});
card.style.transform = scale(${scale});
card.style.boxShadow = shadow;
});
requestAnimationFrame(animateDepth);
}

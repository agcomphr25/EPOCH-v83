Notification Delivery Rules

const allowSms = Boolean(process.env.TWILIO_SID && process.env.TWILIO_AUTH_TOKEN);
const prefersSms = preferredMethods.includes("sms");
const prefersEmail = preferredMethods.includes("email");

let notificationSucceeded = false;
const succeededMethods: string[] = [];

// --- EMAIL PATH ---
if (prefersEmail) {
  try {
    await sendEmailNotification(...);
    succeededMethods.push("email");
    notificationSucceeded = true;
    console.log("[NOTIFY] Email succeeded");
  } catch (err) {
    console.error("[NOTIFY-EMAIL-FAIL]", err);
  }
}

// --- SMS PATH ---
if (prefersSms) {
  if (!allowSms) {
    console.log("[NOTIFY] SMS preferred but Twilio missing → fallback to email only");
  } else {
    try {
      await sendSMSNotification(...);
      succeededMethods.push("sms");
      notificationSucceeded = true;
      console.log("[NOTIFY] SMS succeeded");
    } catch (err) {
      console.error("[NOTIFY-SMS-FAIL]", err);
    }
  }
}

// Final decision
if (!notificationSucceeded) {
  throw new Error("No notification method succeeded");
}

// Update order only if success
await updateOrderNotificationStatus(orderId, succeededMethods);
